Input : Adjacency matrix A[n][n]
Output: Transitive closure matrix T

1. T <- A

2. For k <- 1 to n do
      For i <- 1 to n do
           For j <- 1 to n do
                T[i][j] <- T[i][j] OR
                           (T[i][k] AND T[k][j])
           End For
      End For
   End For

3. Return T
