Input : Weighted graph G(V,E)
Output: Shortest distance from source

1. For each vertex v do
      dist[v] <- infinity
      visited[v] <- FALSE
   End For

2. dist[source] <- 0

3. Repeat n times
      u <- unvisited vertex with minimum dist

      Mark u as visited

      For each neighbor v of u do
           if dist[u] + weight(u,v) < dist[v] then
                dist[v] <- dist[u] + weight(u,v)
           end if
      End For
   End Repeat

4. Return dist[]
