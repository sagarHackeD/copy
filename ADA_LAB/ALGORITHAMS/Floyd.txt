Input : Weight matrix W[n][n]
Output: Shortest distance matrix D

1. D <- W

2. For k <- 1 to n do
      For i <- 1 to n do
           For j <- 1 to n do
                D[i][j] <- min(D[i][j],
                              D[i][k] + D[k][j])
           End For
      End For
   End For

3. Return D
