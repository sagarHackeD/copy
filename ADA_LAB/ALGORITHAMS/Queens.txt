NQUEENS(k, n)
Input : n X n chessboard
Output: All possible solutions

1. For column <- 1 to n do

      if PLACE(k, column) then

           x[k] <- column

           if k = n then
                Print solution
           else
                NQUEENS(k+1, n)
           end if

      end if

   End For
   
PLACE(k, column)
1. For i <- 1 to k-1 do

      if x[i] = column OR
         abs(x[i] - column) =
         abs(i - k) then

           Return FALSE
      end if

   End For

2. Return TRUE
