\begin{tikzpicture}[
level 1/.style={sibling distance=7cm},
level 2/.style={sibling distance=5cm},
level 3/.style={sibling distance=25cm},
every node/.style={circle,draw,align=center}
]

child{node{Gchild2} }
}

};
