18. SQL Queries
(i) Create table statement for Reserves
CREATE TABLE Reserves
(
    sid INTEGER,
    bid INTEGER,
    day DATE,
    PRIMARY KEY (sid, bid, day),
    FOREIGN KEY (sid) REFERENCES Sailors(sid),
    FOREIGN KEY (bid) REFERENCES Boats(bid)
);
(ii) Find all information of sailors who reserved boat 101
SELECT *
FROM Sailors
WHERE sid IN
(
    SELECT sid
    FROM Reserves
    WHERE bid = 101
);
(iii) Names of sailors who reserved at least one boat
SELECT DISTINCT S.sname
FROM Sailors S, Reserves R
WHERE S.sid = R.sid;
(iv) Names of sailors who reserved a red boat
SELECT DISTINCT S.sname
FROM Sailors S, Reserves R, Boats B
WHERE S.sid = R.sid
AND R.bid = B.bid
AND B.color = 'Red';
(v) Average age of sailors for each rating level
SELECT rating,
       AVG(age) AS Average_Age
FROM Sailors
GROUP BY rating;
