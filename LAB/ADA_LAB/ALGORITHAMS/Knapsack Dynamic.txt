Input : Capacity W, weights wt[], values val[]
Output: Maximum profit
1. Create table K[0...n][0...W]
2. For i <- 0 to n do
      For w <- 0 to W do
           if i = 0 or w = 0 then
                K[i][w] <- 0
           else if wt[i] <= w then
                K[i][w] <- max(
                             val[i] +
                             K[i-1][w-wt[i]],
                             K[i-1][w]
                           )
           else
                K[i][w] <- K[i-1][w]
           end if
      End For
   End For
3. Return K[n][W]
