\begin{algorithm}
\caption{Euclid's Algorithm}
\label{euclid}
\begin{algorithmic}[1]
\Require Two non-negative integers $a$ and $b$ such that $a \geq b$
\Ensure $d = \gcd(a,b)$
\State $r \gets a \bmod b$
\While{$r \neq 0$}
\State $a \gets b$
\State $b \gets r$
\State $r \gets a \bmod b$
\EndWhile
\State \Return $b$
\end{algorithmic}
\end{algorithm}
