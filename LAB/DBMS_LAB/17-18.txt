(i) Colors of boats reserved by Alber 
SELECT DISTINCT B.color FROM Sailors S, Boats B, Reserves R WHERE S.sid = R.sid AND B.bid = R.bid AND S.sname = 'Alber'; 
(ii) Sailor IDs with rating ≥ 8 or reserved boat 103 
SELECT sid FROM Sailors WHERE rating >= 8 UNION SELECT sid FROM Reserves WHERE bid = 103;
(iii) Names of sailors who have not reserved a boat containing "storm" 
SELECT sname FROM Sailors WHERE sid NOT IN ( SELECT R.sid FROM Reserves R, Boats B WHERE R.bid = B.bid AND B.bname LIKE '%storm%' ) ORDER BY sname ASC;
(iv) Sailor IDs with age > 20 and not reserved a boat containing "thunder" 
SELECT sid FROM Sailors WHERE age > 20 AND sid NOT IN ( SELECT R.sid FROM Reserves R, Boats B WHERE R.bid = B.bid AND B.bname LIKE '%thunder%' );

(i) Create table statement for Reserves 
CREATE TABLE Reserves ( sid INTEGER, bid INTEGER, day DATE, PRIMARY KEY (sid, bid, day), FOREIGN KEY (sid) REFERENCES Sailors(sid), FOREIGN KEY (bid) REFERENCES Boats(bid) );
(ii) Find all information of sailors who reserved boat 101 
SELECT * FROM Sailors WHERE sid IN ( SELECT sid FROM Reserves WHERE bid = 101 );
(iii) Names of sailors who reserved at least one boat 
SELECT DISTINCT S.sname FROM Sailors S, Reserves R WHERE S.sid = R.sid;
(iv) Names of sailors who reserved a red boat 
SELECT DISTINCT S.sname FROM Sailors S, Reserves R, Boats B WHERE S.sid = R.sid AND R.bid = B.bid AND B.color = 'Red';
(v) Average age of sailors for each rating level 
SELECT rating, AVG(age) AS Average_Age FROM Sailors GROUP BY rating;
