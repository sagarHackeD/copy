-- User Management

GRANT ALL PRIVILEGES ON *.* TO 'user'@'localhost';
FLUSH PRIVILEGES;

-- Table Operations (DDL)
CREATE TABLE Employee (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    salary DECIMAL(10,2)
);

ALTER TABLE Employee ADD age INT;
ALTER TABLE Employee MODIFY name VARCHAR(100);
ALTER TABLE Employee RENAME COLUMN name TO emp_name;
DROP TABLE Employee;

-- Data Operations (DML)
INSERT INTO Employee VALUES (1, 'A', 50000, 25);
UPDATE Employee SET salary = 60000 WHERE id = 1;
DELETE FROM Employee WHERE id = 1;

-- Transaction Control (TCL)
SET autocommit = 0;
ROLLBACK;

-- Basic Queries
SELECT * FROM Employee;
SELECT COUNT(*) FROM Employee;
SELECT MAX(salary) FROM Employee;
SELECT MIN(salary) FROM Employee;

-- Sorting
SELECT * FROM Employee ORDER BY salary ASC;

-- Grouping
SELECT age, SUM(salary)
FROM Employee
GROUP BY age;

-- Joins
SELECT *
FROM Sailors S, Boats B, Reserves R
WHERE S.sid = R.sid
AND B.bid = R.bid;

-- DISTINCT
SELECT DISTINCT color FROM Boats;

-- UNION
SELECT sid FROM Sailors
UNION
SELECT sid FROM Reserves;

-- Subquery
SELECT *
FROM Sailors
WHERE sid IN (
    SELECT sid
    FROM Reserves
);

-- NOT IN
SELECT *
FROM Sailors
WHERE sid NOT IN (
    SELECT sid
    FROM Reserves
);

-- LIKE
SELECT *
FROM Boats
WHERE bname LIKE '%storm%';

-- Aggregate + GROUP BY
SELECT rating, AVG(age)
FROM Sailors
GROUP BY rating;
