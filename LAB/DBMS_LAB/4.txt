>DELIMITER // 
>CREATE TRIGGER salary_after_insert AFTER INSERT ON CUSTOMERS FOR EACH ROW BEGIN INSERT INTO salary_log (action_type, old_salary, new_salary, difference) VALUES ('INSERT', 0, NEW.SALARY, NEW.SALARY - 0); END// 
>DELIMITER ;
>CREATE TRIGGER salary_after_update AFTER UPDATE ON CUSTOMERS FOR EACH ROW BEGIN INSERT INTO salary_log (action_type, old_salary, new_salary, difference) VALUES ('UPDATE', OLD.SALARY, NEW.SALARY, NEW.SALARY - OLD.SALARY); END//  
>CREATE TRIGGER salary_after_delete AFTER DELETE ON CUSTOMERS FOR EACH ROW BEGIN INSERT INTO salary_log (action_type, old_salary, new_salary, difference) VALUES ('DELETE', OLD.SALARY, 0, 0 - OLD.SALARY); END// 
>CREATE TABLE salary_log ( id INT AUTO_INCREMENT PRIMARY KEY, action_type VARCHAR(20), old_salary INT, new_salary INT, difference INT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP );
