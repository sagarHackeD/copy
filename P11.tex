\begin{algorithm}
\caption{Euclid's Algorithm for GCD}
\begin{algorithmic}[1]
\Require Two non-negative integers $a$ and $b$ such that $a \ge b$
\Ensure Returns $\gcd(a,b)$
\State $r \gets a \bmod b$
\While{$r \neq 0$}
\State $a \gets b$
\State $b \gets r$
\State $r \gets a \bmod b$
\EndWhile
\State \Return $b$
\end{algorithmic}
\end{algorithm}
